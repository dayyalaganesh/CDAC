# Paste:

CREATE DATABASE IF NOT EXISTS telecom_db;

USE telecom_db;

CREATE EXTERNAL TABLE IF NOT EXISTS telecom_usage_raw (
    usage_id STRING,
    customer_id STRING,
    plan STRING,
    call_minutes INT,
    sms_count INT,
    data_gb DOUBLE,
    usage_date DATE,
    circle STRING
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/usecases/telecom_customer_usage/raw'
TBLPROPERTIES ("skip.header.line.count"="1");

# Paste:

USE telecom_db;

DROP TABLE IF EXISTS telecom_usage_final;

CREATE TABLE telecom_usage_final AS
SELECT
    usage_id,
    customer_id,
    plan,
    call_minutes,
    sms_count,
    data_gb,
    usage_date,
    circle,
    CASE
        WHEN data_gb >= 10 THEN 'HIGH'
        ELSE 'NORMAL'
    END AS usage_category
FROM telecom_usage_raw;

# Paste:

from airflow import DAG
from airflow.operators.bash import BashOperator
from datetime import datetime

with DAG(
    dag_id="telecom_usage_etl",
    start_date=datetime(2026, 10, 5),
    schedule=None,
    catchup=False
) as dag:

    upload_to_hdfs = BashOperator(
        task_id="upload_to_hdfs",
        bash_command="""
        hdfs dfs -mkdir -p /usecases/telecom_customer_usage/raw

        if hdfs dfs -test -e /usecases/telecom_customer_usage/raw/telecom_usage.csv
        then
            echo "File already exists in HDFS"
        else
            hdfs dfs -put /home/ganesh/usecases/telecom_customer_usage/input/telecom_usage.csv /usecases/telecom_customer_usage/raw/
        fi
        """,
    )

    create_raw_table = BashOperator(
        task_id="create_raw_table",
        bash_command="""
        beeline -u 'jdbc:hive2://localhost:10001' \
        -f /home/ganesh/usecases/telecom_customer_usage/sql/01_create_telecom_raw_table.sql
        """,
    )

    create_final_table = BashOperator(
        task_id="create_final_table",
        bash_command="""
        beeline -u 'jdbc:hive2://localhost:10001' \
        -f /home/ganesh/usecases/telecom_customer_usage/sql/02_create_telecom_final_table.sql
        """,
    )

    upload_to_hdfs >> create_raw_table >> create_final_table
